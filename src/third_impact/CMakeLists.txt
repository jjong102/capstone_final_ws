cmake_minimum_required(VERSION 3.8)
project(third_impact)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rviz_common REQUIRED)
find_package(pluginlib REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
set(CMAKE_AUTOMOC ON)
add_library(third_impact_panel SHARED cpp/goal_panel.cpp include/third_impact/goal_panel.hpp)
target_include_directories(third_impact_panel PRIVATE include)
target_link_libraries(third_impact_panel Qt5::Widgets)
ament_target_dependencies(third_impact_panel rclcpp std_msgs rviz_common pluginlib)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS third_impact_panel LIBRARY DESTINATION lib)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/impact_node scripts/debug_recorder scripts/localization_bootstrap DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config rviz maps DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(core test/test_core.py)
  ament_add_pytest_test(supervisor test/test_supervisor.py)
  ament_add_pytest_test(localization_search test/test_localization_search.py TIMEOUT 90)
  ament_add_pytest_test(localization_policy test/test_localization_policy.py)
  ament_add_pytest_test(runtime test/test_runtime.py)
  add_executable(panel_smoke test/panel_smoke.cpp)
  target_link_libraries(panel_smoke Qt5::Widgets)
  ament_target_dependencies(panel_smoke rclcpp std_msgs pluginlib rviz_common)
  find_package(nav2_core REQUIRED)
  find_package(rclcpp_lifecycle REQUIRED)
  add_executable(patrol_checker_smoke test/patrol_checker_smoke.cpp)
  ament_target_dependencies(patrol_checker_smoke rclcpp rclcpp_lifecycle pluginlib nav2_core)
endif()
ament_package()
